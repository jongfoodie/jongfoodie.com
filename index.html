/** * Jong Foodie — Social Auto-Post Cloud Functions * * Deze functies draaien op de server (Firebase Cloud Functions), niet in de browser. * Dat is expres: de Instagram/TikTok tokens mogen NOOIT in client-side code staan, * anders kan iedereen die de broncode van je site bekijkt ze stelen. * * VOORWAARDE: je hebt Deel A uit de handleiding afgerond (goedgekeurde apps + tokens), * anders geven deze functies een foutmelding terug. * * DEPLOY: * 1. npm install -g firebase-tools (als je dat nog niet hebt) * 2. cd naar de map met dit bestand * 3. firebase login * 4. firebase init functions (kies je bestaande "jongfoodie" project) * 5. npm install (in de functions map) * 6. firebase deploy --only functions */ const functions = require('firebase-functions'); const admin = require('firebase-admin'); const fetch = require('node-fetch'); admin.initializeApp(); const db = admin.firestore(); // ── Helper: haal de opgeslagen tokens op uit Firestore ────────────────────── // Deze staan in het document settings/socialTokens, ingevuld via het // instellingenscherm in admin.html. Alleen deze server-functie leest ze uit — // niet de browser/website zelf. async function getSocialTokens() { const snap = await db.collection('settings').doc('socialTokens').get(); if (!snap.exists) { throw new Error('Nog geen social media tokens ingesteld. Ga naar admin.html → Social Instellingen.'); } return snap.data(); } // ── INSTAGRAM ──────────────────────────────────────────────────────────────── // Gebruikt de Instagram Graph API (content publishing). // Documentatie: https://developers.facebook.com/docs/instagram-api/guides/content-publishing exports.postToInstagram = functions.https.onCall(async (data, context) => { // Simpele beveiliging: alleen toestaan als het verzoek een geheime sleutel meestuurt // die je zelf instelt (zie ADMIN_SECRET hieronder). Vervang dit voor productie // liever door Firebase Auth + admin-check. const { imageUrl, caption, adminSecret } = data; const expectedSecret = functions.config().jongfoodie?.admin_secret; if (!expectedSecret || adminSecret !== expectedSecret) { throw new functions.https.HttpsError('permission-denied', 'Ongeldige admin-sleutel.'); } if (!imageUrl) { throw new functions.https.HttpsError('invalid-argument', 'imageUrl is verplicht — de foto moet al een publieke URL hebben (bv. Firebase Storage link).'); } const tokens = await getSocialTokens(); const { igBusinessAccountId, igAccessToken } = tokens; if (!igBusinessAccountId || !igAccessToken) { throw new functions.https.HttpsError('failed-precondition', 'Instagram is nog niet gekoppeld. Vul je tokens in bij Social Instellingen.'); } try { // Stap 1: maak een media-container aan const containerRes = await fetch( `https://graph.facebook.com/v19.0/${igBusinessAccountId}/media`, { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ image_url: imageUrl, caption: caption || '', access_token: igAccessToken }) } ); const containerData = await containerRes.json(); if (containerData.error) { throw new Error(`Instagram container error: ${containerData.error.message}`); } // Stap 2: publiceer de container const publishRes = await fetch( `https://graph.facebook.com/v19.0/${igBusinessAccountId}/media_publish`, { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ creation_id: containerData.id, access_token: igAccessToken }) } ); const publishData = await publishRes.json(); if (publishData.error) { throw new Error(`Instagram publish error: ${publishData.error.message}`); } return { success: true, postId: publishData.id }; } catch (err) { console.error('Instagram post failed:', err); throw new functions.https.HttpsError('internal', err.message); } }); // ── TIKTOK ─────────────────────────────────────────────────────────────────── // Gebruikt de TikTok Content Posting API (video upload + publish). // Documentatie: https://developers.tiktok.com/doc/content-posting-api-get-started exports.postToTikTok = functions.https.onCall(async (data, context) => { const { videoUrl, caption, adminSecret } = data; const expectedSecret = functions.config().jongfoodie?.admin_secret; if (!expectedSecret || adminSecret !== expectedSecret) { throw new functions.https.HttpsError('permission-denied', 'Ongeldige admin-sleutel.'); } if (!videoUrl) { throw new functions.https.HttpsError('invalid-argument', 'videoUrl is verplicht.'); } const tokens = await getSocialTokens(); const { tiktokAccessToken, tiktokOpenId } = tokens; if (!tiktokAccessToken) { throw new functions.https.HttpsError('failed-precondition', 'TikTok is nog niet gekoppeld. Vul je tokens in bij Social Instellingen.'); } try { // Stap 1: init de upload (TikTok haalt de video op via een publieke URL — PULL_FROM_URL) const initRes = await fetch('https://open.tiktokapis.com/v2/post/publish/video/init/', { method: 'POST', headers: { 'Authorization': `Bearer ${tiktokAccessToken}`, 'Content-Type': 'application/json' }, body: JSON.stringify({ post_info: { title: caption || '', privacy_level: 'PUBLIC_TO_EVERYONE', disable_duet: false, disable_comment: false, disable_stitch: false }, source_info: { source: 'PULL_FROM_URL', video_url: videoUrl } }) }); const initData = await initRes.json(); if (initData.error && initData.error.code !== 'ok') { throw new Error(`TikTok init error: ${initData.error.message}`); } return { success: true, publishId: initData.data?.publish_id }; } catch (err) { console.error('TikTok post failed:', err); throw new functions.https.HttpsError('internal', err.message); } }); // ── OAUTH CODE EXCHANGE — Instagram ───────────────────────────────────────── // Called right after the admin redirects back from Facebook's login dialog // with a one-time "code". This swaps that code for a real access token, // finds the linked Instagram Business Account, and saves everything to // Firestore automatically — no manual copy/paste needed. exports.exchangeInstagramCode = functions.https.onCall(async (data, context) => { const { code, appId, redirectUri, adminSecret } = data; const expectedSecret = functions.config().jongfoodie?.admin_secret; if (!expectedSecret || adminSecret !== expectedSecret) { throw new functions.https.HttpsError('permission-denied', 'Ongeldige admin-sleutel.'); } const appSecret = functions.config().meta?.app_secret; if (!appSecret) { throw new functions.https.HttpsError('failed-precondition', 'Meta App Secret is nog niet ingesteld op de server. Draai: firebase functions:config:set meta.app_secret="JOUW_SECRET"'); } try { // Step 1: exchange the code for a short-lived user access token const tokenRes = await fetch( `https://graph.facebook.com/v19.0/oauth/access_token?client_id=${encodeURIComponent(appId)}&redirect_uri=${encodeURIComponent(redirectUri)}&client_secret=${encodeURIComponent(appSecret)}&code=${encodeURIComponent(code)}` ); const tokenData = await tokenRes.json(); if (tokenData.error) throw new Error('Token exchange failed: ' + tokenData.error.message); const shortLivedToken = tokenData.access_token; // Step 2: exchange for a long-lived token (valid ~60 days) const longRes = await fetch( `https://graph.facebook.com/v19.0/oauth/access_token?grant_type=fb_exchange_token&client_id=${encodeURIComponent(appId)}&client_secret=${encodeURIComponent(appSecret)}&fb_exchange_token=${encodeURIComponent(shortLivedToken)}` ); const longData = await longRes.json(); if (longData.error) throw new Error('Long-lived token exchange failed: ' + longData.error.message); const longLivedToken = longData.access_token; // Step 3: find the user's Facebook Pages, then the Instagram Business Account linked to the first one const pagesRes = await fetch(`https://graph.facebook.com/v19.0/me/accounts?access_token=${encodeURIComponent(longLivedToken)}`); const pagesData = await pagesRes.json(); if (pagesData.error) throw new Error('Could not fetch pages: ' + pagesData.error.message); if (!pagesData.data || pagesData.data.length === 0) { throw new Error('No Facebook Page found on this account. Make sure your Instagram Business account is linked to a Facebook Page.'); } const page = pagesData.data[0]; const pageAccessToken = page.access_token; const igRes = await fetch(`https://graph.facebook.com/v19.0/${page.id}?fields=instagram_business_account&access_token=${encodeURIComponent(pageAccessToken)}`); const igData = await igRes.json(); if (igData.error) throw new Error('Could not fetch Instagram account: ' + igData.error.message); if (!igData.instagram_business_account) { throw new Error('This Facebook Page has no linked Instagram Business account.'); } const igBusinessAccountId = igData.instagram_business_account.id; // Step 4: save everything to Firestore, ready for postToInstagram to use await db.collection('settings').doc('socialTokens').set({ igBusinessAccountId, igAccessToken: pageAccessToken, igConnectedAt: admin.firestore.FieldValue.serverTimestamp() }, { merge: true }); return { success: true, igBusinessAccountId }; } catch (err) { console.error('Instagram OAuth exchange failed:', err); throw new functions.https.HttpsError('internal', err.message); } }); // ── OAUTH CODE EXCHANGE — TikTok ──────────────────────────────────────────── exports.exchangeTikTokCode = functions.https.onCall(async (data, context) => { const { code, clientKey, redirectUri, adminSecret } = data; const expectedSecret = functions.config().jongfoodie?.admin_secret; if (!expectedSecret || adminSecret !== expectedSecret) { throw new functions.https.HttpsError('permission-denied', 'Ongeldige admin-sleutel.'); } const clientSecret = functions.config().tiktok?.client_secret; if (!clientSecret) { throw new functions.https.HttpsError('failed-precondition', 'TikTok Client Secret is nog niet ingesteld op de server. Draai: firebase functions:config:set tiktok.client_secret="JOUW_SECRET"'); } try { const tokenRes = await fetch('https://open.tiktokapis.com/v2/oauth/token/', { method: 'POST', headers: { 'Content-Type': 'application/x-www-form-urlencoded' }, body: new URLSearchParams({ client_key: clientKey, client_secret: clientSecret, code, grant_type: 'authorization_code', redirect_uri: redirectUri }) }); const tokenData = await tokenRes.json(); if (tokenData.error) throw new Error('TikTok token exchange failed: ' + (tokenData.error_description || tokenData.error)); await db.collection('settings').doc('socialTokens').set({ tiktokAccessToken: tokenData.access_token, tiktokOpenId: tokenData.open_id, tiktokRefreshToken: tokenData.refresh_token || '', tiktokConnectedAt: admin.firestore.FieldValue.serverTimestamp() }, { merge: true }); return { success: true, openId: tokenData.open_id }; } catch (err) { console.error('TikTok OAuth exchange failed:', err); throw new functions.https.HttpsError('internal', err.message); } }); // ── Instelling van je admin-sleutel (eenmalig, via terminal) ──────────────── // firebase functions:config:set jongfoodie.admin_secret="KIES_EEN_LANG_WACHTWOORD" // firebase functions:config:set meta.app_secret="JOUW_META_APP_SECRET" // firebase functions:config:set tiktok.client_secret="JOUW_TIKTOK_CLIENT_SECRET"
